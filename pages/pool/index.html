Connect your miner

Any stratum endpoint below — pick the closest region. TLS optional where noted.
# Stratum (plain TCP)
stratum+tcp://pool.example.com:3333
# Miner login: your prl1... address, worker after the dot
-u prl1YOURADDRESS.worker1 -p x
# Solo mining: prefix the wallet with solo:
-u solo:prl1YOURADDRESS.worker1 -p x
# Static difficulty (Kryptex-style): -p d=4194304

spark-pearl-miner

# pool preset (TOML)
[pools.foundry]
url = "stratum+tcp://pool.example.com:3333"
wallet = "prl1YOURADDRESS"
worker = "rig1"

miner-cpm90hx (CMP 90HX)

# object authorize dialect, plain_proof field
./miner --pool stratum+tcp://pool.example.com:3333 \
  --wallet prl1YOURADDRESS --worker cmp1
